279class LikelihoodCalculator:
280    """개인식별지수(Likelihood Ratio) 계산 전담 클래스
281
282    STR 프로필의 allele 빈도를 기반으로 개인식별지수를 계산합니다.
283    allele_frequency 데이터는 클래스 레벨에서 한 번만 로드됩니다.
284    """
285
286    _allele_frequency: pd.DataFrame | None = None
287
288    @classmethod
289    def _load_frequency_data(cls) -> pd.DataFrame:
290        """allele frequency 데이터를 로드 (최초 1회만 실행)"""
291        if cls._allele_frequency is None:
292            module_dir = os.path.dirname(os.path.abspath(__file__))
293            csv_path = os.path.join(module_dir, "allele_frequency.csv")
294            logger.debug(f"allele_frequency.csv 로딩 중: {csv_path}")
295            cls._allele_frequency = pd.read_csv(csv_path)
296            logger.info(f"allele_frequency 로드 완료 (rows={len(cls._allele_frequency)})")
297        return cls._allele_frequency
298
299    def __init__(self, kit: str):
300        """
301        Args:
302            kit: 사용할 마커 키트 (STR, STR20, YSTR)
303        """
304        self.kit = kit
305        self.markers = DICT_MARKERS[kit]
306        self.allele_frequency = self._load_frequency_data()
307        self.prob_nomatch = PROB_NOMATCH
308
309    def calculate(self, profile: pd.Series) -> tuple:
310        """프로필의 개인식별지수를 계산하여 반환
311
312        Args:
313            profile: 프로필 데이터 (Series with marker columns)
314
315        Returns:
316            tuple: (mantissa, exponent) 형태의 개인식별지수
317                   예: ("4.0", "10") -> 4.0 x 10^10
318        """
319        logger.debug(f"개인식별지수 계산 시작 (kit={self.kit})")
320        prob_match = 1.0
321
322        for marker in self.markers:
323            if marker not in profile.index:
324                continue
325            if profile[marker] != "":
326                alleles = str(profile[marker]).split("-")
327                frequencies = []
328
329                for allele in alleles:
330                    try:
331                        float_allele = float(allele)
332                        cond1 = self.allele_frequency["Loci"] == marker
333                        cond2 = self.allele_frequency["Allele"] == float_allele
334                        df_frequency = self.allele_frequency[cond1 & cond2]
335                        if df_frequency.shape[0]:
336                            frequencies.append(df_frequency.iloc[0]["Frequency"])
337                        else:
338                            frequencies.append(self.prob_nomatch)
339                    except ValueError:
340                        logger.warning(f"좌위 {marker}에 유효하지 않은 allele - 계산에서 제외 (value={profile[marker]})")
341                        break
342                else:
343                    # break 없이 정상 완료된 경우만 계산
344                    if len(frequencies) == 1:
345                        prob_match = prob_match / (frequencies[0] * frequencies[0])
346                    elif len(frequencies) == 2:
347                        prob_match = prob_match / (2 * frequencies[0] * frequencies[1])
348
349        # 과학적 표기법으로 변환 (예: 4.08e+10 -> "4.0", "10")
350        text_prob = "{0:.2e}".format(prob_match)  # '4.08e+10'
351        text_prob = text_prob[:3] + text_prob[4:]  # '4.0e+10' (버림 연산)
352        result = tuple(text_prob.split("e+"))
353
354        logger.info(f"개인식별지수 계산 완료 (LR={result[0]}x10^{result[1]})")
355        return result

402        series_profile = self.df_profile_sets[self.df_profile_sets["감정물번호"] == code_evidence].iloc[0]
403
404        for locus in self.markers:
405            alleles = self._transform_set_to_list(set(series_profile[locus]))
406            temp_alleles = []
407            if len(alleles) == 0:
408                temp_alleles.append("NC")
409                flag_NC = True
410            elif alleles[0] == "ND":
411                temp_alleles.append("ND")
412                flag_ND = True
413            elif alleles[0] == "NC":
414                temp_alleles.append("NC")
415                flag_NC = True
416            else:
417                for allele in alleles:
418                    if allele.find("OL") != -1:
419                        continue

421                    if allele.find(".") != -1:
422                        if not self._check_special_case(locus, allele):
423                            cnt_microvariant = cnt_microvariant + 1
424                            decimal_place = allele.split(".")[1]
425                            if decimal_place == "1":
426                                modified_allele = str(int(float(allele)))

443                    alleles = alleles * 2
444                string_profile[locus] = "-".join([str(element) for element in alleles])
445                # 특수 케이스

448                if locus == "DYS385":
449                    string_profile[locus] = string_profile[locus].replace("-", ", ")
450        str_etc = str_etc + str_etc_microvariant
451        if flag_ND:
452            str_etc.append("ND : 디엔에이형이 검출되지 않음.")
453        if flag_NC:
454            str_etc.append("NC : 디엔에이형을 결정할 수 없음.")

457    def _check_mixture(self, series_profile: pd.Series) -> bool:
458        """현재 작업중인 프로필이 혼합 프로필인지 검사
459
460        Args:
461            series_profile: 프로필 데이터 시리즈
462
463        Returns:
464            bool: 혼합 프로필 여부
465        """
466        limit_allele = 1 if self.is_ystr else 2
467        cnt_ta = 0
468        for marker in self.markers:
469            alleles = series_profile[marker]
470            cnt_ta = cnt_ta + 1 if len(alleles) > limit_allele else cnt_ta

123    def _filter_evidence_by_ids(self, list_id: list, kit: str) -> pd.DataFrame:
124        """
125        감정물번호 리스트로 증거물 데이터 필터링
126
127        Args:
128            list_id: 필터링할 감정물번호 리스트
129            kit: 키트 종류 ("STR" or "YSTR")
130
131        Returns:
132            필터링 및 전처리된 DataFrame
133        """
134        logger.debug(f"증거물 필터링 시작 (kit={kit}, 입력 ID 수={len(list_id)})")
135        kit_data = self.switch_kit[kit]
136        df = self.report_data.evidenceinfo
137
138        # 1단계: 미기재 제외
139        original_size = len(df)
140        df = df[df[kit_data["colname_text_evidence"]] != "미기재"].copy()
141        logger.debug(f"미기재 제외 (전: {original_size}, 후: {len(df)})")
142
143        # 2단계: 표기번호 컬럼 추가
144        df[&#x27;표기번호'] = df[kit_data["colname_text_evidence"]]
145
146        # 3단계: 다음 감정물번호 추가 (list_id 필터링 전에 계산)
147        # 이렇게 해야 원본 데이터셋에서의 실제 연속성을 정확히 판단할 수 있음
148        # 예: [1,2,3,4,5]에서 4가 미기재면 -> [1,2,3,5]가 되고, 3의 next는 5가 됨
149        # 하지만 list_id=[1,2,3,5,6]으로 필터하기 전에 계산하면 3의 next는 4가 됨
150        # 그래서 3->5는 연속이 아님을 감지할 수 있음
151        df["감정물번호_다음"] = df["감정물번호"].shift(-1)
152
153        # 4단계: 지정된 감정물번호만 필터링
154        df = df[df["감정물번호"].isin(list_id)]
155        logger.debug(f"ID 필터링 후 데이터 크기: {len(df)}")
156
157        return df.reset_index(drop=True)

159    def _add_reaction_info(self, df: pd.DataFrame) -> tuple[pd.DataFrame, str]:
160        """
161        체액 반응 정보를 DataFrame에 추가
162
163        Args:
164            df: 증거물 DataFrame
165
166        Returns:
167            (반응 정보가 추가된 DataFrame, 동일 반응 문자열)
168        """
169        logger.debug(f"체액 반응 정보 처리 시작 (데이터 수={len(df)})")
170        # 각 반응 유형 처리 (순수 함수 사용)
171        reactions = []
172        for col in ["타액_반응", "정액_반응", "혈흔_반응"]:
173            reactions.append(df[col].apply(lambda x, c=col: etu.format_single_reaction(c, x)))
174
175        # 반응 결과 결합
176        combined = reactions[0] + "," + reactions[1] + "," + reactions[2]
177
178        # 빈 문자열 제거 및 괄호 추가 (순수 함수 사용)
179        df["반응실험결과"] = combined.apply(etu.format_reaction_parentheses)
180
181        # 모두 같은 반응이면 한 번만 표시 (순수 함수 사용)
182        equivalent_reaction = etu.check_equivalent_reaction(df)
183        if equivalent_reaction:
184            logger.debug(f"동일 반응 감지: {equivalent_reaction}")
185            df["반응실험결과"] = ""
186
187        # 표기번호에 반응 추가
188        df["표기번호"] = df["표기번호"] + df["반응실험결과"]
189        logger.debug("체액 반응 정보 처리 완료")
190
191        return df, equivalent_reaction

193    def _create_text_evidence(self, list_id: list, kit: str = "STR", reaction: bool = False) -> str:
194        """
195        증거물 번호를 감정서 형식으로 변환
196
197        연속된 감정물번호는 ~로 묶어 표현하며, 체액 반응 정보를 포함할 수 있습니다.
198        기존 123줄 메서드를 6개의 작은 헬퍼 메서드로 분리하여 리팩토링했습니다.
199
200        개선사항:
201        - 단일 책임 원칙 적용
202        - 복잡도 감소 (13+ → 5)
203        - 테스트 가능한 작은 단위로 분리
204        - 명확한 메서드명으로 의도 전달
205
206        Args:
207            list_id: 증거물 번호 리스트
208            kit: 키트 종류 ("STR" or "YSTR")
209            reaction: 체액 반응 포함 여부
210
211        Returns:
212            포맷팅된 증거물 문자열 (예: "증1호~증3호")
213
214        Examples:
215            >>> writer._create_text_evidence(["2023-D-1-1", "2023-D-1-2", "2023-D-1-3"])
216            "증1호~증3호"
217
218            >>> writer._create_text_evidence(["2023-D-1-1"], reaction=True)
219            "증1호(타액반응 양성)"
220        """
221        logger.debug(f"증거물 텍스트 생성 시작 (kit={kit}, 증거물 수={len(list_id)}, reaction={reaction})")
222
223        # 1. 데이터 필터링
224        df = self._filter_evidence_by_ids(list_id, kit)
225
226        # 2. 특수 케이스 처리
227        if len(df) == 0:
228            logger.debug("빈 데이터프레임 (결과: 빈 문자열)")
229            return ""
230        if len(df) == 1:
231            # 단일 증거물
232            logger.debug("단일 증거물 처리")
233            if reaction:
234                df, equiv = self._add_reaction_info(df)
235                result = df.iloc[0]["표기번호"] + equiv
236                logger.debug(f"증거물 텍스트 생성 완료: {result}")
237                return result
238            result = df.iloc[0]["표기번호"]
239            logger.debug(f"증거물 텍스트 생성 완료: {result}")
240            return result
241        if len(df) == 2:
242            # 두 개 증거물: "및"로 연결
243            logger.debug("2개 증거물 처리 (&#x27;및'로 연결)")
244            if reaction:
245                df, equiv = self._add_reaction_info(df)
246            result = f"{df.iloc[0][&#x27;표기번호']} 및 {df.iloc[1]['표기번호']}"
247            logger.debug(f"증거물 텍스트 생성 완료: {result}")
248            return result
249
250        # 3. 체액 반응 처리
251        logger.debug(f"3개 이상 증거물 처리 (총 {len(df)}개)")
252        equivalent_reaction = ""
253        if reaction:
254            df, equivalent_reaction = self._add_reaction_info(df)
255
256        # 4. 연속 번호 그룹화 (순수 함수 사용)
257        chains = etu.group_consecutive_evidence(df)
258
259        # 5. 최종 문자열 생성 (순수 함수 사용)
260        result = etu.format_evidence_text(chains, equivalent_reaction)
261        logger.debug(f"증거물 텍스트 생성 완료: {result}")
262        return result

355    def make_contents_with_profile(self, phraser: Callable, type_profile:Literal["대표", "대조"], kit:Literal["STR", "YSTR"]="STR") -> None:
356        """
357            감정서에 들어갈 프로필이 존재하는 증거물(대표, 대조, 일치)에 대한 결과 문구를 작성
358        """
359        logger.info(f"프로필 문구 생성 시작 (type_profile={type_profile}, kit={kit})")
360        var_kit = self.switch_kit[kit]
361
362        if type_profile not in var_kit["code_categorized"].keys():
363            logger.info(f"{type_profile} 프로필이 분류 결과에 없습니다 (건너뜀)")
364            return
365
366        logger.info(f"{type_profile} 프로필 {len(var_kit[&#x27;code_categorized'][type_profile])}개 발견")
367        for idx, code in enumerate(var_kit["code_categorized"][type_profile], 1):
368            logger.info(f"[{idx}/{len(var_kit[&#x27;code_categorized'][type_profile])}] 프로필 처리 중 (code={code}, type={type_profile})")
369
370            # 1. 참조 정보 및 일치 정보 가져오기
371            info_ref = var_kit[&#x27;info_indexed'].loc[[(code, type_profile)], :]
372            logger.debug(f"참조 정보 행 수: {len(info_ref) if isinstance(info_ref, pd.DataFrame) else 1}")
373            info_match = self._get_match_info(var_kit, code)
374
375            id_ref = info_ref.iloc[0]["감정물번호"]
376            nickname_ref = str(info_ref.iloc[0][var_kit["colname_nickname"]])
377            logger.info(f"참조 프로필 정보 (id={id_ref}, nickname={nickname_ref})")
378
379            # 2. 타입별 처리
380            if type_profile == "대조":
381                self._create_reference_block(info_ref, id_ref, nickname_ref, var_kit, kit)
382            elif type_profile == "대표":
383                info_match = self._merge_representative_and_match(info_ref, info_match)
384
385            # 3. 프로필 속성 추출 및 문구/블록 생성
386            gender, lr = self._extract_profile_properties(kit, id_ref)
